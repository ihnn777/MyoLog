    function openExerciseLogging(exerciseName) {

      appState.currentExercise = exerciseName;

      document.getElementById('modalTitle').textContent = exerciseName;

      

      const backBtn = document.getElementById('modalBackBtn');

      backBtn.style.display = 'block';

      backBtn.textContent = `← ${appState.currentMuscle || 'Back'}`;


      const exerciseLogs = appState.logs.filter(l => l.exerciseName === exerciseName);

      const lastLog = exerciseLogs.length > 0 ? exerciseLogs[exerciseLogs.length - 1] : null;


      // CONTEXT-SPECIFIC INCREMENTS LOGIC

      const isSmall = appState.currentMuscle && EXERCISE_DB[appState.currentMuscle] 

        ? EXERCISE_DB[appState.currentMuscle].isSmall 

        : true;

      

      // Small groups (Shoulders, Arms, Core, Calves): 0.5kg / 1kg / 2kg

      // Large groups (Chest, Quads, Back, Glutes, Hamstrings): 2.5kg / 5kg / 10kg

      const increments = isSmall ? [-2, -1, 0.5, 1, 2] : [-10, -2.5, 2.5, 5, 10];


      // Hip Thrust defaults to Machine variation if no prior entry exists

      let defaultVariation = "Barbell";

      if (lastLog) {

        defaultVariation = lastLog.variation;

      } else if (exerciseName.toLowerCase().includes("hip thrust")) {

        defaultVariation = "Machine";

      }


      const body = document.getElementById('modalBody');

      body.innerHTML = `

        <div class="quick-ref-box">

          <span>Last Logged:</span>

          <strong>${lastLog ? `${lastLog.weight}${lastLog.unit} × ${lastLog.reps} (${lastLog.date})` : 'No past logs'}</strong>

        </div>


        <div class="graph-card">

          <canvas id="progressionChart"></canvas>

        </div>


        <div class="variation-select-group">

          <select id="varSelect" class="select-style" onchange="renderGraph()">

            ${VARIATIONS.map(v => `<option value="${v}" ${defaultVariation === v ? 'selected' : ''}>${v}</option>`).join('')}

          </select>

        </div>


        <div class="logging-form">

          <div class="input-row">

            <div class="input-group">

              <label class="input-label">Weight</label>

              <div class="field-with-unit">

                <input type="number" id="weightInput" class="num-input" value="${lastLog ? lastLog.weight : 20}" step="0.5">

                <select id="unitSelect" class="unit-select">

                  <option value="kg" ${lastLog && lastLog.unit === 'kg' ? 'selected' : ''}>kg</option>

                  <option value="lb" ${lastLog && lastLog.unit === 'lb' ? 'selected' : ''}>lb</option>

                </select>

              </div>

            </div>

            <div class="input-group">

              <label class="input-label">Reps</label>

              <input type="text" id="repsInput" class="num-input" value="${lastLog ? lastLog.reps : '8-10'}">

            </div>

          </div>


          <div class="increment-chips">

            ${increments.map(inc => `<button class="chip-btn" onclick="adjustWeight(${inc})">${inc > 0 ? '+' : ''}${inc}kg</button>`).join('')}

          </div>


          <div class="input-group">

            <label class="input-label">Date</label>

            <input type="date" id="dateInput" class="num-input" value="${new Date().toISOString().split('T')[0]}">

      openModal();

      renderGraph();

      renderHistoryBox();

    }


    function adjustWeight(amount) {

      const input = document.getElementById('weightInput');

      let val = parseFloat(input.value) || 0;

      val = Math.max(0, val + amount);

      input.value = val;

    }

    }

    function deleteLog(id) {

      appState.logs = appState.logs.filter(l => l.id !== id);

      renderHistoryBox();

      renderRecentChips();

    }

    function openWorkoutHistory() {

      const container = document.getElementById('historyModalBody');

      const grouped = {};

      appState.logs.forEach(log => {

        if (!grouped[log.date]) grouped[log.date] = [];

        grouped[log.date].push(log);

      });


      const dates = Object.keys(grouped).sort().reverse();


      if (dates.length === 0) {

        container.innerHTML = '<div style="color:var(--modal-subtext); text-align:center;">No completed workouts logged yet.</div>';

      } else {

        container.innerHTML = dates.map(date => `

            ${grouped[date].map(item => `

              <div style="display:flex; justify-space-between; font-size:0.85rem; padding:4px 0;">

    /* --- BACKUP EXPORT & IMPORT --- */

    function exportData() {

      if (appState.logs.length === 0) {

        alert("No workout data logged yet to back up!");

        return;

      }

      const dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify(appState.logs));

      const downloadAnchor = document.createElement('a');

      downloadAnchor.setAttribute("href", dataStr);

      downloadAnchor.setAttribute("download", `myolog_backup_${new Date().toISOString().split('T')[0]}.json`);

      document.body.appendChild(downloadAnchor);

      downloadAnchor.click();

      downloadAnchor.remove();


      appState.lastBackupCount = appState.logs.length;

      localStorage.setItem('myolog_backup_count', appState.lastBackupCount.toString());

      checkBackupPrompt();

    }


    function triggerImport() {

      document.getElementById('importFileInput').click();

    }


    function importData(event) {

      const file = event.target.files[0];

      if (!file) return;


      const reader = new FileReader();

      reader.onload = function(e) {

        try {

          const importedLogs = JSON.parse(e.target.result);

          if (Array.isArray(importedLogs)) {

            appState.logs = importedLogs;

            localStorage.setItem('myolog_logs', JSON.stringify(appState.logs));

            renderRecentChips();

            alert("Backup restored successfully!");

          } else {

            alert("Invalid backup file format.");

          }

        } catch (err) {

          alert("Error reading file: " + err.message);

        }

      };

      reader.readAsText(file);

    }
